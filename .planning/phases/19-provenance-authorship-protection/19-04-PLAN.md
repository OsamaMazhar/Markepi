<objective>
Build an evaluation harness and production gate for invisible creator-protection watermarking.

**Purpose:** Decide responsibly whether an open-source invisible watermark can ship in Markepi. The default product protection already comes from C2PA/IPTC; invisible watermarking only ships if it protects creators without damaging image quality, HDR, metadata, performance, or licensing.

**Output:** Provider protocol, local evaluation documentation/harness, license notices, TrustMark evaluation, InvisMark research note, and a clear ship/defer decision. A provider can ship only if it runs inside the iOS app and Share Extension without a desktop helper, Python/PyTorch runtime, required cloud call, or incompatible model/package license. Current verification says C2PA Swift is production-iOS viable, while TrustMark and InvisMark are evaluation/research candidates until proven otherwise.
</objective>

## Reference Implementation

### `InvisibleWatermarkProvider.swift` (Task 1)

```swift
import Foundation

public struct InvisibleWatermarkOptions: Sendable {
    public var preserveHDR: Bool = true
    public init() {}
}

public struct InvisibleWatermarkResult: Sendable {
    public let outputURL: URL
    /// True only if a detect()-after-embed roundtrip confirmed the payload.
    /// A provider may NOT report success without this (D-22 honesty gate).
    public let verifiedAfterEmbed: Bool
}

public struct InvisibleWatermarkDetection: Sendable {
    public let found: Bool
    public let payload: Data?
}

public enum InvisibleWatermarkError: Error, Sendable, Equatable {
    case unavailable          // no vetted provider installed
    case unsupportedMedia
    case embedFailed
}

/// Boundary for any FUTURE invisible-watermark provider (D-08/D-21/D-22).
/// Kept out of the rendering engine until a provider passes the ship gate.
/// Marks are creator-protection / soft-binding — never authenticity proof (IW-04).
public protocol InvisibleWatermarkProvider: Sendable {
    var providerID: String { get }
    var displayName: String { get }
    var licenseNotice: String { get }        // MIT/Apache notice text to bundle
    var isAvailable: Bool { get }            // false until a vetted provider ships
    var supportsStillImages: Bool { get }
    var supportsVideo: Bool { get }
    var supportsHDR: Bool { get }
    func embed(payload: Data, into source: URL,
               options: InvisibleWatermarkOptions) async throws -> InvisibleWatermarkResult
    func detect(from source: URL) async throws -> InvisibleWatermarkDetection
}

/// Always-unavailable default so UI/receipts can say "off"/"unavailable"
/// honestly without implying a provider exists (D-08, D-22).
public struct NoopInvisibleWatermarkProvider: InvisibleWatermarkProvider {
    public init() {}
    public var providerID: String { "noop" }
    public var displayName: String { "None" }
    public var licenseNotice: String { "" }
    public var isAvailable: Bool { false }
    public var supportsStillImages: Bool { false }
    public var supportsVideo: Bool { false }
    public var supportsHDR: Bool { false }

    public func embed(payload: Data, into source: URL,
                      options: InvisibleWatermarkOptions) async throws -> InvisibleWatermarkResult {
        throw InvisibleWatermarkError.unavailable
    }
    public func detect(from source: URL) async throws -> InvisibleWatermarkDetection {
        InvisibleWatermarkDetection(found: false, payload: nil)
    }
}
```

Tests (Swift Testing):
```swift
import Testing
@testable import WatermarkCore

@Suite("InvisibleWatermarkProvider")
struct InvisibleWatermarkProviderTests {
    @Test("Noop provider is unavailable and safe as the default")
    func noopIsUnavailable() async {
        let provider = NoopInvisibleWatermarkProvider()
        #expect(provider.isAvailable == false)
        await #expect(throws: InvisibleWatermarkError.unavailable) {
            _ = try await provider.embed(payload: Data(), into: URL(fileURLWithPath: "/dev/null"),
                                         options: InvisibleWatermarkOptions())
        }
    }
    @Test("Noop detect reports nothing found")
    func noopDetectsNothing() async throws {
        let d = try await NoopInvisibleWatermarkProvider().detect(from: URL(fileURLWithPath: "/dev/null"))
        #expect(d.found == false)
    }
}
```
