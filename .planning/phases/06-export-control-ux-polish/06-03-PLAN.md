Output: Video progress bar UI, cancelable export pipeline, background notification support, iOS 18 AVFoundation API migration.

       a. If `onProgress` is provided, use structured concurrency:
          ```swift
          try await withThrowingDiscardingTaskGroup { group in
              // Task 1: iterate states for progress
              let states = exportSession.states(updateInterval: 0.1)
              group.addTask {
                  var startTime = Date()
                  for await state in states {
                      switch state {
                      case .exporting(let progress):
                          let fraction = progress.fractionCompleted
                          let elapsed = Date().timeIntervalSince(startTime)
                          let eta: TimeInterval? = fraction > 0.01
                              ? elapsed / max(fraction, 0.01) - elapsed
                              : nil
                          onProgress?(fraction, eta)
                      case .completed:
                          break
                      case .cancelled:
                          throw PipelineError.videoCancelled
                      case .failed(let error):
                          throw PipelineError.videoExportFailed(error)
                      default:
                          break
                      }
                  }
              }
              // Task 2: execute export
              group.addTask {
                  try await exportSession.export(to: outputURL, as: exportSession.outputFileType ?? .mp4)
              }
          }
          ```
          The ETA calculation per D-11: `elapsedTime / max(progress, 0.01) - elapsedTime`. Display as nil (shown as "--") when progress < 0.01.

          ```swift
          let states = exportSession.states(updateInterval: 0.1)

          }
          try await exportSession.export(to: outputURL, as: exportSession.outputFileType ?? .mp4)
          ```
          Even without progress reporting, we use the new API to avoid deprecation warnings.
